#!/usr/bin/env bash # # expand - Ubuntu Filesystem Expansion Tool (LVM and plain partitions) # https://github.com/SolubleSpork/expand # # Usage: curl -sL solublespork.github.io/expand | sudo bash # set -euo pipefail # version.json is the source of truth -- keep this in sync with it on every release. VERSION="2.1.0-beta" DRY_RUN=false AUTO_MODE=false HAVE_LVM=false # Colors RED='\033[0;31m' GREEN='\033[0;32m' YELLOW='\033[1;33m' BLUE='\033[0;34m' CYAN='\033[0;36m' DIM='\033[2m' BOLD='\033[1m' NC='\033[0m' # Global array to store scan results declare -a EXPANDABLE_ITEMS=() # Unallocated space at the end of a disk that growpart can't reach, keyed by # the VG it would be added to (see find_new_pv_space). declare -A NEWPV_DISK=() declare -A NEWPV_BYTES=() # Warnings collected during the scan, printed after it finishes. declare -a SCAN_NOTES=() print_success() { echo -e "${GREEN}✓${NC} $1"; } print_error() { echo -e "${RED}✗${NC} $1"; } print_warning() { echo -e "${YELLOW}!${NC} $1"; } confirm() { local response echo -en "${YELLOW}?${NC} $1 [Y/n] " read -r response < /dev/tty [[ ! "$response" =~ ^[Nn]$ ]] } run_cmd() { local cmd="$1" if $DRY_RUN; then echo -e " ${DIM}[dry-run]${NC} $cmd" else echo -e " ${DIM}Running:${NC} $cmd" eval "$cmd" fi } check_root() { if [[ $EUID -ne 0 ]]; then print_error "This script must be run as root (use sudo)" exit 1 fi } check_lvm() { if command -v pvs &> /dev/null; then HAVE_LVM=true else HAVE_LVM=false fi } check_growpart() { if ! command -v growpart &> /dev/null; then print_error "growpart not found. Install with: apt install cloud-guest-utils" exit 1 fi } is_supported_fstype() { case "$1" in ext4|ext3|ext2|xfs|btrfs) return 0 ;; *) return 1 ;; esac } # Disks already rescanned this run, so a disk with multiple partitions # (e.g. sda2 + sda3 both on sda) only gets rescanned once. declare -a RESCANNED_DISKS=() # NOTE: every exit path below is an explicit `return 0`. These are called as # bare statements in loops under `set -e` -- a function that implicitly # returns non-zero (e.g. via a bare `return` right after a failed `[[ ]]`, or # via the exit status of its last `&&`/`||` branch) would silently kill the # whole script the moment that non-zero path is taken, which for growpart's # very common "nothing left to grow" case is most runs after the first. rescan_disk_once() { local disk_base="$1" if [[ " ${RESCANNED_DISKS[*]} " == *" $disk_base "* ]]; then return 0 fi # Best-effort: not all controllers expose this (e.g. virtio-blk); harmless no-op if missing. { echo 1 > "/sys/class/block/$disk_base/device/rescan"; } 2>/dev/null || true RESCANNED_DISKS+=("$disk_base") return 0 } # Grow the partition backing $1 (a partition device, e.g. /dev/sda3) to match # the real disk size. Non-destructive: only rewrites the partition table. grow_partition() { local part_device="$1" part_base disk_base disk partnum growpart_out part_base=$(basename "$part_device") disk_base=$(lsblk -no PKNAME "$part_device" 2>/dev/null | head -1) || true if [[ -z "$disk_base" ]]; then return 0 fi disk="/dev/$disk_base" rescan_disk_once "$disk_base" if [[ ! -f "/sys/class/block/$part_base/partition" ]]; then return 0 fi partnum=$(cat "/sys/class/block/$part_base/partition" 2>/dev/null || echo "") if [[ -z "$partnum" ]]; then return 0 fi if growpart_out=$(growpart "$disk" "$partnum" 2>&1); then print_success "Grew partition $part_device to match disk size" elif [[ "$growpart_out" != *NOCHANGE* ]]; then print_warning "growpart $disk $partnum: $growpart_out" fi return 0 } # Grow partitions (and PVs, where LVM is in use) to match the real (possibly # just-resized-in-Proxmox) disk size, so the free-space scan below sees # accurate numbers whether or not LVM is involved. Non-destructive to # filesystem data: only touches the partition table and PV metadata. Always # runs for real, even in dry-run mode, so scan results stay accurate; # lvextend/filesystem growth remain gated behind dry-run/confirm. sync_storage_layers() { local pv_device pv_list=" " if $HAVE_LVM; then while read -r pv_device; do pv_device=$(echo "$pv_device" | xargs) [[ -z "$pv_device" ]] && continue pv_list+="$pv_device " grow_partition "$pv_device" local before after before=$(pvs --noheadings -o pv_size --units b "$pv_device" 2>/dev/null | xargs) || true pvresize "$pv_device" &> /dev/null || true after=$(pvs --noheadings -o pv_size --units b "$pv_device" 2>/dev/null | xargs) || true [[ -n "$before" && -n "$after" && "$before" != "$after" ]] && print_success "Resized PV $pv_device" done < <(pvs --noheadings -o pv_name 2>/dev/null) fi # Plain (non-LVM) mounted partitions: grow directly, no PV involved. local name fstype mountpoint while read -r name fstype mountpoint; do [[ -z "$name" || -z "$mountpoint" ]] && continue is_supported_fstype "$fstype" || continue [[ "$pv_list" == *" /dev/$name "* ]] && continue grow_partition "/dev/$name" done < <(lsblk -rno NAME,FSTYPE,MOUNTPOINT 2>/dev/null) } # Print " " for a new partition that # would fill the unallocated space after the last partition on $1. partnum is # the lowest unused number, which is what parted assigns on GPT. new_pv_layout() { local disk="$1" ss last_end used n=1 sep="" ss=$(blockdev --getss "$disk" 2>/dev/null || echo 512) last_end=$(partx -g -o END "$disk" 2>/dev/null | sort -n | tail -1 | xargs) || true used=$(partx -g -o NR "$disk" 2>/dev/null | tr -d ' ') || true while grep -qx "$n" <<< "$used"; do n=$((n + 1)) done [[ "$disk" =~ [0-9]$ ]] && sep="p" echo "$(( ((${last_end:-0} + 1) * ss + 1048575) / 1048576 )) $n ${disk}${sep}${n}" return 0 } # Find unallocated space (> 100MB) at the end of a GPT disk holding an LVM PV # whose partition isn't the last one, so growpart can't grow it -- e.g. a # legacy-BIOS install later switched to UEFI, with the EFI partition appended # after the PV. Read-only: results feed scan_system, and the partition is only # created by prepare_new_pv after confirmation. Works out free space from the # real disk size rather than the GPT header, which may not have been moved to # the new end of the disk yet. find_new_pv_space() { NEWPV_DISK=() NEWPV_BYTES=() $HAVE_LVM || return 0 local -A disk_vgs=() disk_pvparts=() local pv_device vg part_base disk_base partnum while read -r pv_device vg; do [[ -z "$pv_device" || -z "$vg" ]] && continue part_base=$(basename "$pv_device") [[ -f "/sys/class/block/$part_base/partition" ]] || continue disk_base=$(lsblk -no PKNAME "$pv_device" 2>/dev/null | head -1) || true [[ -z "$disk_base" ]] && continue partnum=$(cat "/sys/class/block/$part_base/partition" 2>/dev/null || echo "") [[ -z "$partnum" ]] && continue disk_vgs[$disk_base]+="$vg " disk_pvparts[$disk_base]+="$partnum " done < <(pvs --noheadings -o pv_name,vg_name 2>/dev/null) local disk vgs vg_count pttype last_nr ss disk_bytes last_end free_bytes p t missing for disk_base in "${!disk_vgs[@]}"; do disk="/dev/$disk_base" vgs=$(tr ' ' '\n' <<< "${disk_vgs[$disk_base]}" | sed '/^$/d' | sort -u) vg_count=$(wc -l <<< "$vgs") if [[ "$vg_count" -ne 1 ]]; then SCAN_NOTES+=("$disk has PVs from more than one VG; not offering its free space") continue fi pttype=$(blkid -o value -s PTTYPE "$disk" 2>/dev/null) || true [[ "$pttype" == "gpt" ]] || continue # Last partition by position on disk, not by number. read -r last_nr last_end < <(partx -g -o NR,END "$disk" 2>/dev/null | sort -k2 -n | tail -1) || true [[ -z "${last_nr:-}" ]] && continue # A PV that is already the last partition is handled by growpart. [[ " ${disk_pvparts[$disk_base]} " == *" $last_nr "* ]] && continue # Only if growpart really can't grow any PV on this disk. for p in ${disk_pvparts[$disk_base]}; do if growpart -N "$disk" "$p" &> /dev/null; then continue 2 fi done ss=$(blockdev --getss "$disk" 2>/dev/null || echo 512) disk_bytes=$(blockdev --getsize64 "$disk" 2>/dev/null || echo 0) # GPT keeps a 33-sector backup header + table at the very end. free_bytes=$(( disk_bytes - 33 * ss - (last_end + 1) * ss )) (( free_bytes > 104857600 )) || continue missing="" for t in sgdisk parted partprobe; do command -v "$t" &> /dev/null || missing+="$t " done if [[ -n "$missing" ]]; then SCAN_NOTES+=("$disk has $(( free_bytes / 1073741824 ))G unallocated, but ${missing% } not found (apt install gdisk parted)") continue fi NEWPV_DISK[$vgs]="$disk" NEWPV_BYTES[$vgs]="$free_bytes" done return 0 } # Disks already given a new PV this run (so "expand all" with several LVs in # the same VG only creates one partition). declare -a PREPARED_DISKS=() # Turn the unallocated space at the end of $1 into a new PV in VG $2. Creates # a partition, so it only runs after the normal confirm prompt (or in auto # mode) -- never from sync_storage_layers. Under dry-run, run_cmd only prints. prepare_new_pv() { local disk="$1" vg="$2" start_mib partnum new_part actual if [[ " ${PREPARED_DISKS[*]} " == *" $disk "* ]]; then return 0 fi PREPARED_DISKS+=("$disk") read -r start_mib partnum new_part < <(new_pv_layout "$disk") || true # Move the GPT backup header to the new end of the disk first; otherwise # parted -s can't use the new space (it would prompt Fix/Ignore). run_cmd "sgdisk -e $disk" run_cmd "parted -s -a optimal $disk mkpart lvm ${start_mib}MiB 100%" if ! $DRY_RUN; then # Make sure parted gave the partition the number we expect before # setting flags on it. actual=$(parted -sm "$disk" unit B print 2>/dev/null | awk -F: -v s="$(( start_mib * 1048576 ))B" '$2 == s { print $1 }') || true if [[ "$actual" != "$partnum" ]]; then print_error "New partition on $disk is number '${actual:-?}', expected $partnum. Stopping; nothing was added to $vg." exit 1 fi fi run_cmd "parted -s $disk set $partnum lvm on" run_cmd "partprobe $disk" run_cmd "udevadm settle" if ! $DRY_RUN && [[ ! -b "$new_part" ]]; then print_error "$new_part did not appear after partprobe. Stopping; nothing was added to $vg." exit 1 fi run_cmd "pvcreate $new_part" run_cmd "vgextend $vg $new_part" return 0 } show_scan_notes() { local note for note in "${SCAN_NOTES[@]}"; do print_warning "$note" done return 0 } # Convert size string to bytes for comparison size_to_bytes() { local size="$1" local num unit num=$(echo "$size" | sed 's/[^0-9.]//g') unit=$(echo "$size" | sed 's/[0-9.<]//g' | tr '[:lower:]' '[:upper:]') case "$unit" in T|TB|TIB) echo "$num * 1024 * 1024 * 1024 * 1024" | bc 2>/dev/null || echo 0 ;; G|GB|GIB) echo "$num * 1024 * 1024 * 1024" | bc 2>/dev/null || echo 0 ;; M|MB|MIB) echo "$num * 1024 * 1024" | bc 2>/dev/null || echo 0 ;; K|KB|KIB) echo "$num * 1024" | bc 2>/dev/null || echo 0 ;; *) echo "$num" ;; esac } # Main scanning function - finds all expansion opportunities scan_system() { EXPANDABLE_ITEMS=() SCAN_NOTES=() local found_any=false local pv_list=" " if $HAVE_LVM; then find_new_pv_space # Get VG info (free space and total size) declare -A vg_free declare -A vg_size while read -r vg_name free_space total_size; do vg_name=$(echo "$vg_name" | xargs) free_space=$(echo "$free_space" | xargs) total_size=$(echo "$total_size" | xargs) if [[ -n "$vg_name" ]]; then vg_free["$vg_name"]="$free_space" vg_size["$vg_name"]="$total_size" fi done < <(vgs --noheadings -o vg_name,vg_free,vg_size --units g 2>/dev/null) # Scan each LV while read -r lv_line; do [[ -z "$lv_line" ]] && continue local lv_path lv_size vg_name lv_path=$(echo "$lv_line" | awk '{print $1}' | xargs) lv_size=$(echo "$lv_line" | awk '{print $2}' | xargs) vg_name=$(echo "$lv_line" | awk '{print $3}' | xargs) [[ -z "$lv_path" ]] && continue local mountpoint fstype fs_size fs_used fs_pct mountpoint=$(findmnt -n -o TARGET "$lv_path" 2>/dev/null || echo "") fstype=$(blkid -o value -s TYPE "$lv_path" 2>/dev/null || echo "unknown") # Get filesystem usage if mounted if [[ -n "$mountpoint" ]]; then read -r fs_size fs_used fs_pct < <(df -h "$mountpoint" 2>/dev/null | tail -1 | awk '{print $2, $3, $5}') else fs_size="$lv_size" fs_used="-" fs_pct="-" fi # Check if VG has free space local free="${vg_free[$vg_name]:-0}" local free_bytes free_bytes=$(size_to_bytes "$free") # Plus any unallocated disk space that would be added to this VG as a new PV local newpv_disk="${NEWPV_DISK[$vg_name]:-}" local newpv_bytes="${NEWPV_BYTES[$vg_name]:-0}" free_bytes=$(echo "$free_bytes + $newpv_bytes" | bc 2>/dev/null || echo "$free_bytes") # Only show if there's meaningful free space (> 100MB) if (( $(echo "$free_bytes > 104857600" | bc 2>/dev/null || echo 0) )); then found_any=true # Calculate potential size (current + free) local current_num free_num potential_size current_num=$(echo "$fs_size" | sed 's/[^0-9.]//g') free_num=$(echo "$free" | sed 's/[^0-9.]//g') if [[ -n "$newpv_disk" ]]; then free_num=$(echo "scale=2; $free_num + $newpv_bytes / 1073741824" | bc 2>/dev/null || echo "$free_num") fi potential_size=$(echo "$current_num + $free_num" | bc 2>/dev/null || echo "?") # format: lv_path|vg_name|mountpoint|fstype|current_fs_size|potential_size|used|used_pct|newpv_disk # newpv_disk is set when the space first has to be added to the VG as a new PV. EXPANDABLE_ITEMS+=("$lv_path|$vg_name|$mountpoint|$fstype|$fs_size|${potential_size}G|$fs_used|$fs_pct|$newpv_disk") fi done < <(lvs --noheadings -o lv_path,lv_size,vg_name --units g 2>/dev/null) while read -r pv_device; do pv_device=$(echo "$pv_device" | xargs) [[ -n "$pv_device" ]] && pv_list+="$pv_device " done < <(pvs --noheadings -o pv_name 2>/dev/null) fi # Scan plain (non-LVM) mounted partitions with a supported filesystem local name fstype mountpoint while read -r name fstype mountpoint; do [[ -z "$name" || -z "$mountpoint" ]] && continue is_supported_fstype "$fstype" || continue [[ "$pv_list" == *" /dev/$name "* ]] && continue local part_bytes fs_size fs_used fs_pct fs_bytes part_bytes=$(blockdev --getsize64 "/dev/$name" 2>/dev/null || echo 0) read -r fs_size fs_used fs_pct < <(df -h "$mountpoint" 2>/dev/null | tail -1 | awk '{print $2, $3, $5}') fs_bytes=$(size_to_bytes "$fs_size") # Only show if the partition has meaningfully more room than the filesystem (> 100MB) local slack slack=$(echo "$part_bytes - $fs_bytes" | bc 2>/dev/null || echo 0) if (( $(echo "$slack > 104857600" | bc 2>/dev/null || echo 0) )); then found_any=true local potential_size potential_size=$(echo "scale=1; $part_bytes / 1073741824" | bc 2>/dev/null || echo "?") # vg_name left empty to mark this as a plain (non-LVM) partition EXPANDABLE_ITEMS+=("/dev/$name||$mountpoint|$fstype|$fs_size|${potential_size}G|$fs_used|$fs_pct") fi done < <(lsblk -rno NAME,FSTYPE,MOUNTPOINT 2>/dev/null) $found_any } show_scan_results() { local count=${#EXPANDABLE_ITEMS[@]} if [[ $count -eq 0 ]]; then echo "" echo -e "${DIM}No expansion opportunities found.${NC}" echo "" echo "No free space available to expand into. If you recently expanded" echo "the disk in Proxmox, double-check the resize completed there." exit 0 fi echo "" if [[ $count -eq 1 ]]; then echo -e "${GREEN}Found 1 volume that can be expanded:${NC}" else echo -e "${GREEN}Found $count volumes that can be expanded:${NC}" fi echo "" local i=1 for item in "${EXPANDABLE_ITEMS[@]}"; do IFS='|' read -r lv_path vg_name mountpoint fstype current_size potential_size used used_pct newpv_disk <<< "$item" # Calculate how much can be added local current_num potential_num gain current_num=$(echo "$current_size" | sed 's/[^0-9.]//g') potential_num=$(echo "$potential_size" | sed 's/[^0-9.]//g') gain=$(echo "$potential_num - $current_num" | bc 2>/dev/null || echo "?") # Display name: use mountpoint if available, otherwise LV path local display_name="${mountpoint:-$lv_path}" echo -e " ${BOLD}$i)${NC} ${CYAN}$display_name${NC} ${DIM}($fstype)${NC}" echo -e " ${DIM}$lv_path${NC}" echo -e " Filesystem: ${BOLD}$current_size${NC} → could be ${GREEN}$potential_size${NC} ${GREEN}(+${gain}G)${NC}" echo -e " Currently using: $used ($used_pct)" if [[ -n "$newpv_disk" ]]; then echo -e " ${YELLOW}Includes:${NC} add free space on $newpv_disk as new PV to $vg_name" fi echo "" ((i++)) done } show_menu() { local count=${#EXPANDABLE_ITEMS[@]} if [[ $count -gt 1 ]]; then echo -e " ${BOLD}a)${NC} Expand all" fi echo -e " ${BOLD}q)${NC} Quit" echo "" # Build prompt based on count local prompt if [[ $count -eq 1 ]]; then prompt="Select [1/q]: " else prompt="Select [1-$count/a/q]: " fi echo -en "$prompt" read -r choice < /dev/tty case "$choice" in q|Q) echo "Exiting." exit 0 ;; a|A) if [[ $count -gt 1 ]]; then expand_all_lvs else print_error "Invalid input" fi ;; [1-9]*) if (( choice >= 1 && choice <= count )); then expand_single_lv "$((choice - 1))" else print_error "Invalid selection" fi ;; *) print_error "Invalid input" ;; esac } expand_single_lv() { local idx="$1" IFS='|' read -r lv_path vg_name mountpoint fstype current_size potential_size used used_pct newpv_disk <<< "${EXPANDABLE_ITEMS[$idx]}" local display_name="${mountpoint:-$lv_path}" echo "" echo -e "${BOLD}Expanding:${NC} ${CYAN}$display_name${NC}" echo -e "${DIM}$lv_path${NC}" echo "" local step=1 echo "Steps:" if [[ -n "$newpv_disk" ]]; then local start_mib partnum new_part read -r start_mib partnum new_part < <(new_pv_layout "$newpv_disk") || true echo -e " $step. ${DIM}Add free space on $newpv_disk as new PV $new_part to $vg_name${NC}" echo -e " ${DIM}(sgdisk -e, parted mkpart + lvm flag, partprobe, pvcreate, vgextend)${NC}" ((step++)) fi if [[ -n "$vg_name" ]]; then echo -e " $step. ${DIM}lvextend -l +100%FREE $lv_path${NC}" ((step++)) fi case "$fstype" in ext4|ext3|ext2) echo -e " $step. ${DIM}resize2fs $lv_path${NC}" ;; xfs) echo -e " $step. ${DIM}xfs_growfs $mountpoint${NC}" ;; btrfs) echo -e " $step. ${DIM}btrfs filesystem resize max $mountpoint${NC}" ;; esac echo "" if $DRY_RUN || confirm "Proceed?"; then [[ -n "$newpv_disk" ]] && prepare_new_pv "$newpv_disk" "$vg_name" [[ -n "$vg_name" ]] && run_cmd "lvextend -l +100%FREE $lv_path" case "$fstype" in ext4|ext3|ext2) run_cmd "resize2fs $lv_path" ;; xfs) [[ -n "$mountpoint" ]] && run_cmd "xfs_growfs $mountpoint" ;; btrfs) [[ -n "$mountpoint" ]] && run_cmd "btrfs filesystem resize max $mountpoint" ;; esac echo "" if ! $DRY_RUN; then print_success "Expansion complete!" [[ -n "$mountpoint" ]] && df -h "$mountpoint" | tail -1 fi fi } expand_all_lvs() { echo "" echo -e "${BOLD}Expanding all volumes...${NC}" echo "" if ! $DRY_RUN && ! $AUTO_MODE; then if ! confirm "Expand all ${#EXPANDABLE_ITEMS[@]} volumes?"; then echo "Cancelled." return fi fi local step=1 item lv_path vg_name mountpoint fstype newpv_disk needs_newpv=false for item in "${EXPANDABLE_ITEMS[@]}"; do IFS='|' read -r _ _ _ _ _ _ _ _ newpv_disk <<< "$item" if [[ -n "$newpv_disk" ]]; then needs_newpv=true fi done # Add unallocated disk space to its VG as a new PV, where needed if $needs_newpv; then echo -e "${DIM}Step $step: Adding free disk space as new PVs...${NC}" for item in "${EXPANDABLE_ITEMS[@]}"; do IFS='|' read -r _ vg_name _ _ _ _ _ _ newpv_disk <<< "$item" if [[ -n "$newpv_disk" ]]; then prepare_new_pv "$newpv_disk" "$vg_name" fi done echo "" ((step++)) fi # Extend all LVs (plain partitions were already grown by sync_storage_layers) echo -e "${DIM}Step $step: Extending logical volumes...${NC}" for item in "${EXPANDABLE_ITEMS[@]}"; do IFS='|' read -r lv_path vg_name mountpoint fstype _ _ _ _ _ <<< "$item" [[ -n "$vg_name" ]] && run_cmd "lvextend -l +100%FREE $lv_path" done echo "" ((step++)) # Then resize all filesystems echo -e "${DIM}Step $step: Resizing filesystems...${NC}" for item in "${EXPANDABLE_ITEMS[@]}"; do IFS='|' read -r lv_path vg_name mountpoint fstype _ _ _ _ _ <<< "$item" case "$fstype" in ext4|ext3|ext2) run_cmd "resize2fs $lv_path" ;; xfs) [[ -n "$mountpoint" ]] && run_cmd "xfs_growfs $mountpoint" ;; btrfs) [[ -n "$mountpoint" ]] && run_cmd "btrfs filesystem resize max $mountpoint" ;; esac done echo "" if $DRY_RUN; then print_success "Dry-run complete. No changes made." else print_success "All expansions complete!" echo "" echo "New sizes:" df -h --type=ext4 --type=xfs --type=btrfs 2>/dev/null | head -5 || df -h | grep -E '^/dev' | head -5 fi } show_usage() { cat << EOF ${BOLD}expand${NC} v${VERSION} - Filesystem Expansion Tool (LVM and plain partitions) ${BOLD}Usage:${NC} curl -sL solublespork.github.io/expand | sudo bash curl -sL solublespork.github.io/expand | sudo bash -s auto curl -sL solublespork.github.io/expand | sudo bash -s dry-run ${BOLD}Options:${NC} auto Expand everything automatically (no prompts) dry-run Show what would happen without making changes help Show this help ${BOLD}What it does:${NC} 1. growpart (+ pvresize if using LVM) - Grow the partition to match the disk If the PV's partition isn't last (so it can't grow), free space at the end of the disk is offered as a new PV added to the same VG instead 2. lvextend - Extend LV to use free VG space (LVM only) 3. resize2fs/xfs_growfs/btrfs resize - Grow the filesystem EOF } main() { while [[ $# -gt 0 ]]; do case "$1" in auto|--auto|-a) AUTO_MODE=true; shift ;; dry-run|--dry-run|-d) DRY_RUN=true; shift ;; help|--help|-h) show_usage; exit 0 ;; --version|-v) echo "expand v$VERSION"; exit 0 ;; *) print_error "Unknown option: $1"; exit 1 ;; esac done echo "" echo -e "${BOLD}expand${NC} ${DIM}v${VERSION}${NC}" if $DRY_RUN; then echo -e "${YELLOW}(dry-run mode)${NC}" fi check_root check_lvm check_growpart echo -en "\nSyncing partition/LVM layers... " sync_storage_layers echo "done." echo -en "\nScanning... " if scan_system; then echo "done." show_scan_notes show_scan_results if $AUTO_MODE; then expand_all_lvs else show_menu fi else echo "done." show_scan_notes echo "" echo -e "${DIM}No expansion opportunities found.${NC}" echo "" echo "No free space available to expand into. If you recently expanded" echo "the disk in Proxmox, double-check the resize completed there." echo "" fi } main "$@"
